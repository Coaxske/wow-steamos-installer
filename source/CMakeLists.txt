cmake_minimum_required(VERSION 3.21)
project(WoWSteamOS VERSION 0.2.0 LANGUAGES CXX)
option(WOW_ENABLE_DEVELOPER_TOOLS "Build private developer capture/input/check tools" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_SKIP_RPATH TRUE)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Concurrent Widgets)
find_package(SDL2 REQUIRED CONFIG)
find_program(CARGO_EXECUTABLE cargo REQUIRED)
set(WOW_BUILD_MODE public)
set(WOW_RUST_FEATURE_ARGS --no-default-features)
set(WOW_MAIN_QML ui/Main.qml)
if(WOW_ENABLE_DEVELOPER_TOOLS)
  set(WOW_BUILD_MODE developer)
  list(APPEND WOW_RUST_FEATURE_ARGS --features developer-tools)
  find_package(Qt6 6.4 REQUIRED COMPONENTS Test DBus)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(CONTROLLER_DEPS REQUIRED IMPORTED_TARGET gstreamer-1.0 gstreamer-app-1.0 gstreamer-video-1.0 tesseract)
  set(WOW_MAIN_QML ui/DeveloperMain.qml)
  set_source_files_properties(ui/DeveloperMain.qml PROPERTIES QT_RESOURCE_ALIAS "ui/Main.qml")
endif()
set(WOW_RUST_TARGET_DIR ${CMAKE_CURRENT_SOURCE_DIR}/target/${WOW_BUILD_MODE})
set(RUST_LIB ${WOW_RUST_TARGET_DIR}/release/libwow_steamos_core.a)
file(GLOB_RECURSE RUST_SOURCES CONFIGURE_DEPENDS src/*.rs ui/* scripts/* packaging/* tests/* vendor/* .cargo/*)
add_custom_command(OUTPUT ${RUST_LIB}
  COMMAND ${CARGO_EXECUTABLE} build --release --locked --offline --target-dir ${WOW_RUST_TARGET_DIR} ${WOW_RUST_FEATURE_ARGS}
  WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
  DEPENDS ${RUST_SOURCES} Cargo.toml Cargo.lock build.rs build_support.rs CMakeLists.txt LICENSE README.md FEATURES.md VALIDATION.md TEST-ON-DEVICE.md THIRD-PARTY.md
  COMMENT "Building ${WOW_BUILD_MODE} Rust core")
add_custom_target(rust-core DEPENDS ${RUST_LIB})
qt_add_executable(wow-steamos ui/main.cpp ui/Bridge.cpp ui/Bridge.h)
qt_add_resources(wow-steamos "ui" PREFIX "/" FILES ${WOW_MAIN_QML} ui/ActionButton.qml ui/InstallerCheckBox.qml ui/InfoText.qml ui/Guidance.js)
add_dependencies(wow-steamos rust-core)
target_compile_definitions(wow-steamos PRIVATE WOW_ENABLE_DEVELOPER_TOOLS=$<BOOL:${WOW_ENABLE_DEVELOPER_TOOLS}>)
target_link_libraries(wow-steamos PRIVATE ${RUST_LIB} Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Concurrent Qt6::Widgets SDL2::SDL2 pthread dl)
if(WOW_ENABLE_DEVELOPER_TOOLS)
  target_sources(wow-steamos PRIVATE ui/DeveloperMain.cpp ui/ControllerAutomation.cpp ui/ControllerAutomation.h ui/ControllerFrameSafetyChecks.cpp)
  qt_add_resources(wow-steamos "developer-ui" PREFIX "/" FILES ui/TestCenter.qml ui/LiveTests.qml)
  target_link_libraries(wow-steamos PRIVATE Qt6::Test Qt6::DBus PkgConfig::CONTROLLER_DEPS)
  add_executable(controller-geometry-checks ui/ControllerGeometryChecks.cpp)
  target_link_libraries(controller-geometry-checks PRIVATE Qt6::Core Qt6::Qml)
endif()
install(TARGETS wow-steamos RUNTIME DESTINATION bin)
